int findMax(int array[], int size) {
    int max = array[0];

    for (int i = 0; i < size; i++) {
        if (array[i] > max) {
            max = array[i];
        }
    };

    return max;
}

int findMin(int array[], int size) {
    int min = array[0];

    for (int i = 0; i < size; i++) {
        if (array[i] < min) {
            min = array[i];
        };
    };

    return min;
}

    for (int i = 0; i < n; i++) {
        std::cout << arr[i] << " ";
    };
}

void reverseArray(int array[], int size) {
    int left = 0;
    int right = size - 1;

    while (left < right) {
        std::swap(array[left], array[right]);
        left++;
        right--;
    };
}
